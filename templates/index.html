Examples

Quick Start

Works with any HTTP client. Copy and paste.

# AIO Universal Download curl "https://your-domain.com/api/v1/dl?url=https://youtube.com/watch?v=dQw4w9WgXcQ&api=bx_YOUR_KEY" # Spotify Download curl "https://your-domain.com/api/v1/s?url=https://open.spotify.com/track/TRACK_ID&api=bx_YOUR_KEY"
import requests API = "https://your-domain.com/api/v1" key = "bx_YOUR_KEY" params = {"url": "https://youtube.com/watch?v=dQw4w9WgXcQ", "api": key} res = requests.get(f"{API}/dl", params=params) data = res.json() print(data["title"])
const API = 'https://your-domain.com/api/v1' const key = 'bx_YOUR_KEY' async function dl(url) { const params = new URLSearchParams({url, api: key}) const res = await fetch(`${API}/dl?${params}`) const data = await res.json() console.log(data.title) }
